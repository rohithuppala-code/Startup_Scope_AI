

-- 0. Extensions (must be first)
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS vector;

-- ==========================================
-- 1. ENUM TYPES
-- ==========================================
DO $$ BEGIN
CREATE TYPE validation_status AS ENUM ('pending', 'processing', 'completed', 'failed');
EXCEPTION
WHEN duplicate_object THEN null;
END $$;

-- ==========================================
-- 2. CORE TABLE: VALIDATIONS
-- ==========================================
CREATE TABLE IF NOT EXISTS validations (
-- Core Identifiers
id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,

```
-- Input Data
idea_description TEXT NOT NULL,
target_market TEXT,
budget_constraints TEXT,

-- State Management
status validation_status DEFAULT 'pending' NOT NULL,
idempotency_key TEXT NOT NULL,
idea_hash TEXT NOT NULL,

-- Processing Metadata
processing_started_at TIMESTAMPTZ,
retry_count INT DEFAULT 0 CHECK (retry_count >= 0 AND retry_count <= 5),
last_retry_at TIMESTAMPTZ,

-- AI Cost Tracking
model_version VARCHAR(50), 
tokens_used INT DEFAULT 0 CHECK (tokens_used >= 0),
estimated_cost DECIMAL(10, 6) DEFAULT 0.000000 CHECK (estimated_cost >= 0),

-- Output
report_json JSONB,          
markdown_report TEXT,       
error_message TEXT,         

-- Vector Embedding
idea_embedding VECTOR(1536),

-- Timestamps
created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
deleted_at TIMESTAMPTZ,

-- Constraints
CONSTRAINT unique_user_idempotency UNIQUE(user_id, idempotency_key),

-- State Machine Rules
CONSTRAINT completed_requires_output CHECK (
    status != 'completed' OR report_json IS NOT NULL
),
CONSTRAINT failed_requires_error CHECK (
    status != 'failed' OR error_message IS NOT NULL
),
CONSTRAINT processing_requires_timestamp CHECK (
    status != 'processing' OR processing_started_at IS NOT NULL
)
```

);

-- ==========================================
-- 3. PERFORMANCE INDEXES
-- ==========================================

-- Deduplication (prevent duplicate completed ideas)
CREATE UNIQUE INDEX IF NOT EXISTS unique_completed_idea_hash
ON validations(idea_hash)
WHERE status = 'completed' AND deleted_at IS NULL;

-- Queue monitoring
CREATE INDEX IF NOT EXISTS idx_status_created
ON validations(status, created_at);

-- Stuck job detection
CREATE INDEX IF NOT EXISTS idx_processing_timeout
ON validations(processing_started_at)
WHERE status = 'processing';

-- User dashboard
CREATE INDEX IF NOT EXISTS idx_validations_user_created
ON validations(user_id, created_at DESC)
WHERE deleted_at IS NULL;

-- JSON querying
CREATE INDEX IF NOT EXISTS idx_report_json_gin
ON validations USING GIN (report_json);

-- Vector similarity search
CREATE INDEX IF NOT EXISTS idx_idea_embedding
ON validations
USING hnsw (idea_embedding vector_cosine_ops)
WHERE idea_embedding IS NOT NULL;

-- ==========================================
-- 4. AUTO UPDATE TRIGGER
-- ==========================================
CREATE OR REPLACE FUNCTION update_modified_column()
RETURNS TRIGGER AS $$
BEGIN
NEW.updated_at = NOW();
RETURN NEW;
END;
$$ language 'plpgsql';

DROP TRIGGER IF EXISTS update_validations_modtime ON validations;

CREATE TRIGGER update_validations_modtime
BEFORE UPDATE ON validations
FOR EACH ROW
EXECUTE FUNCTION update_modified_column();

-- ==========================================
-- 5. ROW LEVEL SECURITY (RLS)
-- ==========================================
ALTER TABLE validations ENABLE ROW LEVEL SECURITY;

-- SELECT: Users can only see their own data
DROP POLICY IF EXISTS "Users can view own validations" ON validations;
CREATE POLICY "Users can view own validations"
ON validations FOR SELECT
USING (auth.uid() = user_id AND deleted_at IS NULL);

-- INSERT: Users can only insert their own data
DROP POLICY IF EXISTS "Users can insert own validations" ON validations;
CREATE POLICY "Users can insert own validations"
ON validations FOR INSERT
WITH CHECK (auth.uid() = user_id);

-- UPDATE: Block frontend updates (only backend service role bypasses RLS)
DROP POLICY IF EXISTS "Service role can update validations" ON validations;
CREATE POLICY "Service role can update validations"
ON validations FOR UPDATE
USING (false);
